function doGet(e) { return iiHandle_(e); }

function iiHandle_(e) {
  let out;
  try {
    const p = (e && e.parameter) || {};
    if (!p.action) return ContentService.createTextOutput('سيرفر صور الأصناف شغال ✓');
    const data = p.data ? JSON.parse(p.data) : (e.postData && e.postData.contents ? JSON.parse(e.postData.contents) : {});
    const fn = II_ACTIONS[p.action];
    if (!fn) throw new Error('طلب مش معروف');
    out = fn(data) || {};
    out.ok = true;
  } catch (err) {
    const m = String(err && err.message || err);
    out = { ok: false, error: /too many times|مرات كثيرة|urlfetch|Bandwidth quota/i.test(m) ? II_QUOTA_MSG : m };
  } finally {
    iiFlush_();
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

function iiFlush_() {
  if (!__iiN) return;
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(5000);
    const pr = PropertiesService.getScriptProperties(), k = iiDayKey_();
    pr.setProperty(k, String((Number(pr.getProperty(k)) || 0) + __iiN));
    __iiN = 0;
  } catch (e) {} finally { try { lock.releaseLock(); } catch (e) {} }
}

// الدخول: نفس بيانات تاب الموظفين + لازم الاسم موجود في تاب "تطبيق الصور"
function iiUser_(d) {
  const pass = String(d.password || '').trim();
  if (!String(d.email || '').trim() || !pass) throw new Error('اكتب اسمك (أو الإيميل) وكلمة السر');
  const everyone = users_();
  const email = resolveLogin_(d.email, pass, everyone);
  const me = everyone.filter(u => u.email === email && u.pass === pass);
  if (!me.length) throw new Error('الاسم (أو الإيميل) أو كلمة السر غلط');
  if (!me.some(u => u.active)) throw new Error('الحساب ده موقوف - كلم المدير');
  const keys = [normName_(email)].concat(me.map(u => normName_(u.name)).filter(Boolean));
  const sh = iiAccessSheet_();
  const rows = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, 3).getValues() : [];
  const linked = iiLinked_();
  let all = false;
  const want = [];
  rows.forEach(r => {
    if (keys.indexOf(normName_(r[0])) < 0) return;
    if (r[2] === false || String(r[2]).toUpperCase() === 'FALSE') return;
    splitStores_(r[1]).forEach(b => { if (normName_(b) === normName_(ALL)) all = true; else want.push(b); });
  });
  if (!all && !want.length) throw new Error('حسابك مالوش صلاحية على تطبيق الصور - كلم المدير');
  const branches = all ? linked : linked.filter(b => want.some(w => posNormBranch_(w) === posNormBranch_(b)));
  if (!branches.length) throw new Error('الفرع بتاعك مش مربوط ببرنامج البيع - كلم الدعم');
  return { name: me[0].name || email, email: email, branches: branches, all: all };
}
function iiBranch_(d) {

  const b = u.branches.find(x => posNormBranch_(x) === posNormBranch_(d.branch));
  if (!b) throw new Error('الفرع ده مش من صلاحياتك');
  return { u: u, branch: b };
}

// {sh, rows: [[id,name,code,cat,img,seen]], byId: {id: index}}
function iiRead_(branch) {
  const sh = iiSheet_(branch), n = sh.getLastRow() - 1;
  const rows = n > 0 ? sh.getRange(2, 1, n, II_COLS.length).getValues().map(r => r.map(x => String(x == null ? '' : x))) : [];
  const byId = {}, origE = {};
  rows.forEach((r, i) => { if (r[0]) { byId[r[0]] = i; origE[r[0]] = r[4]; } });
  return { sh: sh, rows: rows, byId: byId, origE: origE };
}
// بنكتب الجدول كله مرة واحدة - والصورة اللي حد حفظها في نفس الوقت ما تضيعش
function iiWrite_(t) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const n = t.sh.getLastRow() - 1;
    if (n > 0) {
      t.sh.getRange(2, 1, n, 5).getValues().forEach(r => {
        const id = String(r[0]), e = String(r[4] == null ? '' : r[4]), i = t.byId[id];
        if (i !== undefined && e !== t.origE[id] && t.rows[i] && t.rows[i][0] === id) t.rows[i][4] = e;
      });
    }
    if (t.full && n > t.rows.length) t.sh.getRange(t.rows.length + 2, 1, n - t.rows.length, II_COLS.length).clearContent();
    if (t.rows.length) t.sh.getRange(2, 1, t.rows.length, II_COLS.length).setValues(t.rows);
    t.byId = {}; t.origE = {};
    t.rows.forEach((r, i) => { t.byId[r[0]] = i; t.origE[r[0]] = r[4]; });
  } finally { lock.releaseLock(); }
}

function iiSession_(branch, fresh) {
  const c = CacheService.getScriptCache(), k = iiKey_(branch);
  if (!fresh) {
    try {
      const jar = JSON.parse(c.get(k) || 'null');
      const cfg = iiCfg_(branch);
      if (jar && cfg) return { cfg: cfg, jar: jar, branch: branch };
    } catch (e) {}
  }
  if (!iiCfg_(branch)) throw new Error('الفرع مش مربوط ببرنامج البيع - راجع سطره في تاب "ربط برامج البيع" (الرابط واسم المستخدم وكلمة السر)');
  const s = bsLogin_(branch);
  s.branch = branch;
  c.put(k, JSON.stringify(s.jar), 1200);
  return s;
}
function iiExpired_(res) {
  const code = res.getResponseCode();
  if (code >= 300 && code < 400) return true;
  const t = res.getContentText('UTF-8');
  return /loginform|logincheck\.php/i.test(t.slice(0, 60000)) && !/show_items_cat|edit_item_save|preview_item|id=["']?row\d+/.test(t);
}
function iiReq_(s, path, payload) {
  const r = {
    url: s.cfg.url + path, muteHttpExceptions: true, followRedirects: false,
    headers: { Cookie: Object.keys(s.jar).map(k => k + '=' + s.jar[k]).join('; '), 'User-Agent': 'Mozilla/5.0 (Modaqqeq images)' }
  };
  if (payload) { r.method = 'post'; r.payload = payload; }
  return r;
}
// طلبات كتير مع بعض - لو الجلسة خلصت ندخل تاني مرة واحدة
function iiFetchAll_(s, build) {
  let res = iiUFAll_(build(s));
  if (res.some(iiExpired_)) {
    const s2 = iiSession_(s.branch, true);
    s.jar = s2.jar;
    res = iiUFAll_(build(s));
    if (res.some(iiExpired_)) throw new Error('برنامج البيع بيخرّج الجلسة - جرّب كمان شوية');
  }

}

    res = bsFetch_(s, path, opt);
  }
  return res;
}
// طلب زي اللي المتصفح بيبعته من صفحة تعديل الصنف بالظبط (نفس الكوكيز + Referer + Origin)
function iiPage_(s, path, payload, form) {
  const go = () => {
    const origin = s.cfg.url.replace(/^(https?:\/\/[^/]+).*$/, '$1');
    const r = {
      method: 'post', muteHttpExceptions: true, followRedirects: false, payload: payload,
      headers: {
        Cookie: Object.keys(s.jar).map(k => k + '=' + s.jar[k]).join('; '),
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0 Safari/537.36',
        'X-Requested-With': 'XMLHttpRequest', Referer: s.cfg.url + 'items/new_item.php', Origin: origin,
        Accept: '*/*'
      }
    };
    if (form) r.contentType = 'application/x-www-form-urlencoded; charset=UTF-8';
    const res = iiUF_(s.cfg.url + path, r);
    const sc = res.getAllHeaders()['Set-Cookie'] || res.getAllHeaders()['set-cookie'];
    if (sc) (Array.isArray(sc) ? sc : [sc]).forEach(c => {
      const kv = String(c).split(';')[0], i = kv.indexOf('=');
      if (i > 0) s.jar[kv.slice(0, i).trim()] = kv.slice(i + 1).trim();
    });
    return res;
  };
  return go();
}

// صف من جدول الأصناف في البرنامج ← [id, name, code, cat]
function iiParseRows_(html) {
  const out = [];
  (String(html).match(/<tr[^>]*id=["']?row\d+[\s\S]*?<\/tr>/gi) || []).forEach(tr => {
    const id = (tr.match(/id=["']?row(\d+)/i) || [])[1];
    if (!id) return;
    const pt = (tr.match(/<a[^>]*class=["'][^"']*print_ticket[^>]*>/i) || [])[0] || '';
    let name = pt ? iiAttr_(pt, 'data-name') : '';
    if (!name) name = iiDecode_(((tr.match(/item\.php\?id=\d+["']\s*>([^<]+)<\/a>/i) || [])[1] || '').trim());
    const code = pt ? (iiAttr_(pt, 'data-barcode') || '') : '';
    const cat = iiDecode_(((tr.match(/show_items_cat\(\d+\)["'][^>]*>([^<]*)</i) || [])[1] || '').trim());
    out.push([id, String(name || '').trim(), String(code).trim(), cat]);
  });
  return out;
}
function iiPagesOf_(html) {
  let mx = 1;
  (String(html).match(/current\s*=\s*(\d+)/g) || []).forEach(x => { const n = Number(x.replace(/\D/g, '')); if (n > mx) mx = n; });
  (String(html).match(/search_items\((\d+)\)/g) || []).forEach(x => { const n = Number(x.replace(/\D/g, '')); if (n > mx) mx = n; });
  return mx;
}

// رابط الصورة من صفحة الصنف (item.php) ← "upload/..." أو "-"
function iiImgOf_(html) {
  const m = String(html).match(/<img\b[^>]*class=["'][^"']*preview_item[^"']*["'][^>]*>/i) || String(html).match(/<img\b[^>]*preview_item[^>]*>/i);
  if (m) {
    const src = iiAttr_(m[0], 'src') || '';
    const u = src.match(II_IMG_RE);
    return u ? u[0] : II_NONE;
  }
  return II_NONE;
}

function iiProgress_(t, st) {
  return {
    stage: st ? st.stage : 'none', finished: !!(st && st.stage === 'done'), listed: !!(st && st.fullSeen),
    paused: !(st && st.stage === 'done') && !iiRoom_(30), // التجهيز وقف النهارده علشان رصيد الطلبات
    page: st ? Math.min(st.page || 1, st.pages || st.page || 1) : 0, pages: st ? st.pages || 0 : 0,
    total: t.rows.length, checked: t.rows.filter(r => r[4]).length,
    withImg: t.rows.filter(r => r[4] && r[4] !== II_NONE).length, done: st && st.done || ''
  };
}

// قفل لكل فرع لوحده (علشان تجهيز فرع ما يوقفش حفظ صورة في فرع تاني)
function iiMutex_(key, ttl, waitMs) {
  const c = CacheService.getScriptCache(), end = Date.now() + (waitMs || 0);
  for (;;) {
    const l = LockService.getScriptLock();
    if (l.tryLock(10000)) {
      try {
        if (!c.get(key)) { c.put(key, '1', ttl); return true; }
      } finally { l.releaseLock(); }
    }
    if (Date.now() > end) return false;
    Utilities.sleep(1500);
  }
}
function iiUnmutex_(key) { CacheService.getScriptCache().remove(key); }
function iiMKey_(kind, branch) { return 'iim_' + kind + '_' + Utilities.base64EncodeWebSafe(Utilities.newBlob(posNormBranch_(branch)).getBytes()).slice(0, 150); }

// خطوة تجهيز (حوالي 40 ثانية) - التطبيق بينادي لحد ما finished
// المرحلة 1: صفحات قايمة الأصناف (30 صنف في الصفحة) ← رقم كل صنف واسمه وباركوده وتصنيفه
// المرحلة 2: صفحة كل صنف لسه ما اتعرفتش صورته ← رابط الصورة
// d.restart = يبدأ من الأول (أصناف جديدة / أسماء اتغيرت)
function iiScan_(d) {
  return iiScanCore_(iiBranch_(d).branch, !!d.restart);
}
function iiScanCore_(b, restart) {
  const d = { restart: restart };
  const mk = iiMKey_('scan', b);
  if (!iiMutex_(mk, 90, 0)) return iiProgress_(iiRead_(b), iiScanState_(b)); // فيه حد تاني بيجهّز نفس الفرع
  try {
    const t0 = Date.now(), budget = 40 * 1000;
    let st = iiScanState_(b);
    if (!st || (d.restart && st.stage === 'done')) st = { stage: 'list', page: 1, pages: 0, run: String(Date.now()), started: new Date().toISOString(), fullSeen: st && st.fullSeen || '' };
    const t = iiRead_(b);
    const s = iiSession_(b);
    if (st.stage === 'list') {
      while (Date.now() - t0 < budget && (!st.pages || st.page <= st.pages) && iiRoom_(8)) {
        const batch = [];
        for (let p = st.page; p < st.page + 8 && (!st.pages || p <= st.pages); p++) batch.push(p);
        if (!batch.length) break;
        const res = iiFetchAll_(s, x => batch.map(p => iiReq_(x, 'items/show_items.php', iiListPayload_(p))));
        res.forEach(r => {
          const html = r.getContentText('UTF-8');
          if (!st.pages) st.pages = iiPagesOf_(html);
          iiParseRows_(html).forEach(it => {
            const i = t.byId[it[0]];
            if (i === undefined) { t.byId[it[0]] = t.rows.length; t.rows.push([it[0], it[1], it[2], it[3], '', st.run]); }
            else { const row = t.rows[i]; row[1] = it[1]; row[2] = it[2]; row[3] = it[3]; row[5] = st.run; }
          });
        });
        st.page = batch[batch.length - 1] + 1;
      }
      if (st.pages && st.page > st.pages) {
        // الأصناف اللي اتمسحت من البرنامج: نشيلها
        t.rows = t.rows.filter(r => r[5] === st.run);
        t.byId = {};
        t.rows.forEach((r, i) => { t.byId[r[0]] = i; });
        t.full = true;
        st.fullSeen = st.run;
        st.stage = 'imgs';
      }
      iiWrite_(t);
    }
    if (st.stage === 'imgs' && Date.now() - t0 < budget) {
      const todo = [];
      t.rows.forEach((r, i) => { if (!r[4]) todo.push(i); });
      let k = 0;
      while (Date.now() - t0 < budget && k < todo.length && iiRoom_(30)) {
        const part = todo.slice(k, k + 30);
        const res = iiFetchAll_(s, x => part.map(i => iiReq_(x, 'items/item.php?id=' + t.rows[i][0])));
        part.forEach((i, j) => {
          const html = res[j].getContentText('UTF-8');
          if (res[j].getResponseCode() === 200 && html.length > 1000) t.rows[i][4] = iiImgOf_(html);
        });
        k += part.length;
      }
      iiWrite_(t);
      if (k >= todo.length) { st.stage = 'done'; st.done = new Date().toISOString(); }
    }
    iiSaveJar_(s);
    iiScanSave_(b, st);
    return iiProgress_(t, st);
  } finally {
    iiUnmutex_(mk);
  }
}

// القايمة للتطبيق ← {items: [[id, name, code, cat, img]], base, scan}
function iiItems_(d) {
  const b = iiBranch_(d).branch;
  const cfg = iiCfg_(b);
  if (!cfg) throw new Error('الفرع مش مربوط ببرنامج البيع - راجع سطره في تاب "ربط برامج البيع" (الرابط واسم المستخدم وكلمة السر)');
  const t = iiRead_(b), st = iiScanState_(b);
  const marks = iiMarks_(b);
  // [رقم، اسم، باركود، تصنيف، صورة، نسخة اللوجو اللي على الصورة ('' = من غير لوجو)، ليها نسخة أصلية محفوظة 1/0]
  const items = t.rows.filter(r => r[0]).map(r => {
    const mk = r[4] ? marks[r[0]] : null, on = mk && mk.img === r[4];
    return [r[0], r[1], r[2], r[3], r[4], on ? (mk.ver || 'old') : '', on && mk.orig ? 1 : 0];
  });
  // بصمة القايمة: لو الموبايل عنده نفس النسخة ما نبعتش القايمة تاني
  const hash = Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, cfg.url + JSON.stringify(items), Utilities.Charset.UTF_8));
  if (d.have && d.have === hash) return { same: true, hash: hash, scan: iiProgress_(t, st) };
  return { items: items, base: cfg.url, hash: hash, scan: iiProgress_(t, st) };
}

// ---------- حفظ الصورة ----------
// d = {branch, id, image: "data:image/jpeg;base64,..."}
function iiSet_(d) {
  const ub = iiBranch_(d);
  const m = String(d.image || '').match(/^data:(image\/(?:jpeg|png|webp));base64,(.+)$/);
  if (!m) throw new Error('الصورة مش واضحة - اختار صورة تانية');
  const id = String(d.id || '').replace(/\D/g, '');
  // الصورة عليها لوجو: نحفظ النسخة الأصلية (من غير لوجو) في Drive الأول، علشان لو اللوجو اتغير أو اتشال نرجعلها
  let orig = '', fresh = false;
  if (d.wm && id) {
    try {
      const c = String(d.clean || '').match(/^data:(image\/(?:jpeg|png|webp));base64,(.+)$/);
      if (c) { orig = iiOrigSave_(ub.branch, id, Utilities.newBlob(Utilities.base64Decode(c[2]), c[1])); fresh = true; }
      else if (d.orig && iiOrigOk_(ub.branch, id, String(d.orig))) orig = String(d.orig);
    } catch (e) { orig = ''; }
  }
  let r;
  try {
    r = iiSetCore_(ub.branch, ub.u.name + (d.wm ? ' (بلوجو)' : ''), id, Utilities.newBlob(Utilities.base64Decode(m[2]), m[1]), null);
  } catch (e) {
    if (fresh && orig) { try { DriveApp.getFileById(orig).setTrashed(true); } catch (e2) {} }
    throw e;
  }
  if (d.wm && r.img !== II_NONE) {
    const ver = String(d.wmVer || '').replace(/\D/g, '') || 'old';
    try { iiMark_(ub.branch, r.id, r.img, ver, orig); r.wm = ver; r.orig = orig ? 1 : 0; } catch (e) {}
  }
  return r;
}
// blob = الصورة الجديدة، أو path = رابط صورة موجودة على البرنامج (upload/... - للرجوع بعد التجربة)
function iiSetCore_(b, who, id, blobIn, path) {
  id = String(id || '').replace(/\D/g, '');
  if (!id) throw new Error('الصنف مش معروف');
  const type = blobIn ? blobIn.getContentType() : '';
  const bytes = blobIn ? blobIn.getBytes() : null;
  if (bytes && bytes.length > 3 * 1024 * 1024) throw new Error('الصورة كبيرة قوي');
  const ext = type === 'image/png' ? 'png' : type === 'image/webp' ? 'webp' : 'jpg';
  const mk = iiMKey_('set', b);
  if (!iiRoom_(6, true)) throw new Error(II_QUOTA_MSG);
  if (!iiMutex_(mk, 120, 45000)) throw new Error('فيه صورة تانية بتتحفظ دلوقتي - جرّب تاني');
  let name = '', oldImg = '';
  try {
    const s = iiSession_(b);
    // نفس ترتيب صفحة البرنامج بالظبط: نفتح صفحة تعديل الصنف ← نرفع الصورة ← نحفظ (من غير ما نفتح الصفحة تاني في النص)
    const edit = () => iiFetch_(s, 'items/new_item.php', { method: 'post', payload: { id: id, edit: 'item', store_id: '' } }).getContentText('UTF-8');
    // 1) صفحة تعديل الصنف
    const html = edit();
    if (!/edit_item_save\.php/.test(html)) throw new Error('مقدرتش أفتح صفحة تعديل الصنف في البرنامج');
    const before = iiFormPayload_(html, null);
    if (before.id !== id) throw new Error('البرنامج فتح صنف تاني - ما اتحفظش حاجة');
    name = before.payload.item_name || '';
    oldImg = before.oldImg;
    // 2) رفع الصورة (البرنامج بيرجّع اسم الملف، والصفحة بتحطه upload/الاسم)
    let newImg = path;
    if (bytes) {
      const blob = Utilities.newBlob(bytes, type, 'item_' + id + '.' + ext);
      const up = iiPage_(s, 'uploader.php', { photoimg: blob }, false);
      const raw = String(up.getContentText('UTF-8') || '').trim();
      const file = raw.replace(/^"+|"+$/g, '');
      if (up.getResponseCode() !== 200 || !/^[\w\-\/.]+\.(?:jpe?g|png|gif|webp)$/i.test(file)) {
        iiLog_(b, who, id, name, oldImg, '', 'رفع الصورة فشل', raw.slice(0, 200));
        throw new Error('البرنامج رفض رفع الصورة' + (raw && raw.length < 120 ? ': ' + raw.replace(/<[^>]+>/g, ' ').trim() : ''));
      }
      newImg = 'upload/' + file;
    }
    if (newImg == null) throw new Error('مفيش صورة');
    // 3) الحفظ: نفس بيانات الصفحة بالظبط والصورة الجديدة بس
    const form = iiFormPayload_(html, newImg);
    const sent = iiParam_(form.payload);
    const res = iiPage_(s, 'items/edit_item_save.php', sent, true);
    const msg = String(res.getContentText('UTF-8') || '').trim();
    if (!(Number(msg) > 0)) {
      iiLog_(b, who, id, name, oldImg, newImg, 'الحفظ فشل', msg.slice(0, 300));
      const txt = msg.replace(/<[^>]+>/g, ' ').replace(/\s+/g, ' ').trim();
      throw new Error('البرنامج ما حفظش الصورة' + (txt && txt.length < 150 ? ': ' + txt : ' - غالباً يوزر الربط مالوش صلاحية تعديل الأصناف'));
    }
    // 4) التأكد: نفتح الصنف تاني ونقارن كل الخانات (ما عدا الصورة والكمية لأنها بتتغير مع البيع)
    const after = iiFormPayload_(edit(), null);
    const stored = after.oldImg;
    const diffs = iiDiff_(sent, iiParam_(after.payload));
    const ok = path != null ? (stored === path || (!II_IMG_RE.test(stored) && !II_IMG_RE.test(path))) : (stored !== oldImg && II_IMG_RE.test(stored));
    // 5) شيت الصور
    try { iiSetCell_(b, id, II_IMG_RE.test(stored) ? stored.match(II_IMG_RE)[0] : II_NONE); } catch (e) {}
    iiSaveJar_(s);
    iiLog_(b, who, id, name, oldImg, stored, ok ? (diffs.length ? 'اتحفظت - في خانات اتغيرت' : 'تمام') : 'الصورة ما اتغيرتش',
      (diffs.length ? 'اتغير: ' + diffs.join('، ') : '') + (ok ? '' : ' | اتبعت: ' + newImg));
    if (!ok) throw new Error('البرنامج رد إن الحفظ تمام بس الصورة ما اتغيرتش - كلم الدعم');
    return { img: II_IMG_RE.test(stored) ? stored.match(II_IMG_RE)[0] : II_NONE, url: s.cfg.url + stored, changed: diffs, oldImg: oldImg, name: name, id: id };
  } finally {
    iiUnmutex_(mk);
  }
}

// الخانات اللي اختلفت بين طلبين (ما عدا الصورة والكمية)
function iiDiff_(a, b) {
  const skip = { img: 1, quantity: 1, old_quantity: 1 };
  const A = {}, B = {}, out = [];
  String(a).split('&').forEach(x => { const k = decodeURIComponent(x.split('=')[0]); if (!skip[k]) A[x] = k; });
  String(b).split('&').forEach(x => { const k = decodeURIComponent(x.split('=')[0]); if (!skip[k]) B[x] = k; });
  Object.keys(A).forEach(x => { if (!B[x] && out.indexOf(A[x]) < 0) out.push(A[x]); });
  Object.keys(B).forEach(x => { if (!A[x] && out.indexOf(B[x]) < 0) out.push(B[x]); });
  return out;
}

// ---------- قراءة صفحة تعديل الصنف (نفس اللي الصفحة بتبعته بالظبط) ----------
function iiDecode_(s) {
  const named = { quot: '"', apos: "'", lt: '<', gt: '>', amp: '&', nbsp: '\u00a0' };
  return String(s == null ? '' : s).replace(/&(#\d+|#x[0-9a-f]+|quot|apos|lt|gt|amp|nbsp|#039);/gi, (m, e) => {
    if (e[0] === '#') return String.fromCharCode(e[1] === 'x' || e[1] === 'X' ? parseInt(e.slice(2), 16) : Number(e.slice(1)));
    return named[e.toLowerCase()];
  });
}
function iiAttr_(tag, name) {
  const m = String(tag).match(new RegExp('\\s' + name + '\\s*=\\s*(?:"([^"]*)"|\'([^\']*)\'|([^\\s>]+))', 'i'));
  return m ? iiDecode_(m[1] != null ? m[1] : (m[2] != null ? m[2] : m[3])) : null;
}
function iiHasAttr_(tag, name) { return new RegExp('\\s' + name + '(?=[\\s=/>])', 'i').test(String(tag)); }
// كل التاجات من نوع معين (من غير اللي جوه السكريبتات)
function iiTags_(html, tagName) {

}

// زي اللي $.post بيبعته بالظبط: الخانة اللي مش موجودة في الصفحة (undefined) ما بتتبعتش خالص
// (لو اتبعتت فاضية البرنامج بيرجّع "تمام" بس ما بيحفظش الصورة)، null = فاضي، المصفوفات [0][key]
function iiParam_(obj) {
  const out = [];
  const add = (k, v) => out.push(encodeURIComponent(k) + '=' + encodeURIComponent(v == null ? '' : v));
  const build = (prefix, v) => {
    if (v === undefined) return;
    if (Array.isArray(v)) v.forEach((x, i) => build(prefix + '[' + (typeof x === 'object' && x != null ? i : '') + ']', x));
    else if (v != null && typeof v === 'object') Object.keys(v).forEach(k => build(prefix + '[' + k + ']', v[k]));
    else add(prefix, v);
  };
  Object.keys(obj).forEach(k => build(k, obj[k]));
  return out.join('&');
}

// d = {branch, ids: [...]} ← {imgs: {id: "upload/..." أو "-"}}
function iiPeek_(d) {
  const b = iiBranch_(d).branch;
  const t = iiRead_(b);
  const ids = (Array.isArray(d.ids) ? d.ids : []).map(x => String(x).replace(/\D/g, ''))
    .filter((x, i, a) => x && a.indexOf(x) === i && t.byId[x] !== undefined && !t.rows[t.byId[x]][4]).slice(0, 30);
  const imgs = {};
  if (!ids.length) return { imgs: imgs };
  if (!iiRoom_(ids.length, true)) throw new Error(II_QUOTA_MSG);
  const s = iiSession_(b);
  const res = iiFetchAll_(s, x => ids.map(id => iiReq_(x, 'items/item.php?id=' + id)));
  ids.forEach((id, j) => {
    const html = res[j].getContentText('UTF-8');
    if (res[j].getResponseCode() === 200 && html.length > 1000) { imgs[id] = iiImgOf_(html); t.rows[t.byId[id]][4] = imgs[id]; }
  });
  if (Object.keys(imgs).length) iiWrite_(t);
  iiSaveJar_(s);
  return { imgs: imgs };
}

function posBaseUrl_(v) {
  let u = String(v || '').trim();
  if (!u) return '';
  if (!/^https?:\/\//i.test(u)) u = 'https://' + u;
  u = u.replace(/[?#].*$/, '').replace(/\/(login|index|logincheck)\.php$/i, '/');
  if (!/\/$/.test(u)) u += '/';
  return u;
}
function posNormBranch_(x) {
  return String(x || '').replace(/[ىی]/g, 'ي').replace(/ة/g, 'ه').replace(/[أإآ]/g, 'ا').replace(/\s+/g, ' ').trim();
}
// {url, user, pass} أو null لو الفرع مش مربوط
function posCfgFromSheet_(branch) {
  const b = posNormBranch_(branch);
  if (!b) return null;
  const cache = CacheService.getScriptCache();
  const ck = 'poscfg_' + Utilities.base64EncodeWebSafe(Utilities.newBlob(b).getBytes()).slice(0, 200);
  const hit = cache.get(ck);
  if (hit) return hit === '-' ? null : JSON.parse(hit);
  const sh = posLinksSheet_();
  let cfg = null;
  if (sh.getLastRow() > 1) {
    const rows = sh.getRange(2, 1, sh.getLastRow() - 1, 4).getValues();
    for (const r of rows) {
      if (posNormBranch_(r[0]) !== b) continue;
      const url = posBaseUrl_(r[1]);
      const user = String(r[2] || '').trim(), pass = String(r[3] || '');
      if (!url || !user || !pass) break;
      cfg = { url: url, user: user, pass: pass };
      break;
    }
  }
  cache.put(ck, cfg ? JSON.stringify(cfg) : '-', 300);
  return cfg;
}
function bsLogin_(branch) {
  const cfg = posCfgFromSheet_(branch);
  if (!cfg) throw new Error('الفرع مش مربوط ببرنامج البيع');
  const s = { cfg: cfg, jar: {} };
  bsFetch_(s, 'login.php?lan=ar');
  const r = bsFetch_(s, 'logincheck.php', { method: 'post', ajax: true, payload: { username: cfg.user, password: cfg.pass, lang: 'ar' } });
  const msg = r.getContentText('UTF-8').trim();
  if (!(Number(msg) > 0)) throw new Error('مقدرتش أدخل على برنامج البيع: ' + (msg.length > 200 ? 'رابط البرنامج غلط' : (msg.replace(/<[^>]+>/g, ' ').trim() || 'رد فاضي')));
  return s;
}
function bsFetch_(s, path, opt) {
  opt = opt || {};
  let url = s.cfg.url + path;
  for (let hop = 0; hop < 5; hop++) {
    const headers = { 'User-Agent': 'Mozilla/5.0 (Modaqqeq images)' };
    const ck = Object.keys(s.jar).map(k => k + '=' + s.jar[k]).join('; ');
    if (ck) headers.Cookie = ck;
    if (opt.ajax) headers['X-Requested-With'] = 'XMLHttpRequest';
    const req = { method: hop === 0 ? (opt.method || 'get') : 'get', headers: headers, followRedirects: false, muteHttpExceptions: true };
    if (hop === 0 && opt.payload) req.payload = opt.payload;
    if (hop === 0 && opt.form) req.contentType = 'application/x-www-form-urlencoded; charset=UTF-8';
    const res = iiUF_(url, req);
    const all = res.getAllHeaders();
    const sc = all['Set-Cookie'] || all['set-cookie'];
    if (sc) (Array.isArray(sc) ? sc : [sc]).forEach(c => {
      const kv = String(c).split(';')[0], i = kv.indexOf('=');
      if (i > 0) s.jar[kv.slice(0, i).trim()] = kv.slice(i + 1).trim();
    });
    const loc = all['Location'] || all['location'];
    if (res.getResponseCode() >= 300 && res.getResponseCode() < 400 && loc) {
      url = /^https?:/i.test(loc) ? loc : (loc.charAt(0) === '/' ? s.cfg.url.replace(/^(https?:\/\/[^/]+).*$/, '$1') + loc : s.cfg.url + loc);
      continue;
    }
    return res;
  }
  throw new Error('برنامج البيع عمل تحويلات كتير');
}

// الموظفين (نفس تاب الدخول): الإيميل | كلمة السر | الاسم | الفروع | نشط ... | الموبايل
function users_() {
  const sh = iiDirSS_().getSheetByName(SH_USERS);
  if (!sh || sh.getLastRow() < 2) return [];
  const rows = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getValues();
  const ph = rows[0].map(x => String(x).trim()).indexOf('الموبايل');
  return rows.slice(1).map(r => ({
    email: String(r[0]).trim().toLowerCase(), pass: String(r[1]).trim(), name: String(r[2]).trim(),
    active: isActive_(r[4]), phone: ph >= 0 ? String(r[ph] || '').trim() : ''
  })).filter(u => u.email);
}
function isActive_(v) { return !(v === false || String(v).toUpperCase() === 'FALSE'); }
function splitStores_(v) { return String(v || '').split(/[,،]/).map(x => x.trim()).filter(Boolean); }
function normName_(s) {
  return String(s || '').trim().replace(/\s+/g, ' ').toLowerCase()
    .replace(/[أإآ]/g, 'ا').replace(/ى/g, 'ي').replace(/ة/g, 'ه');
}
function digits_(v) {
  return String(v == null ? '' : v).replace(/[٠-٩]/g, d => '٠١٢٣٤٥٦٧٨٩'.indexOf(d)).replace(/[۰-۹]/g, d => '۰۱۲۳۴۵۶۷۸۹'.indexOf(d)).replace(/[^0-9]/g, '');
}
function phoneKey_(v) {
  let n = digits_(v);
  if (n.indexOf('0020') === 0) n = n.slice(4);
  else if (n.indexOf('20') === 0 && n.length === 12) n = n.slice(2);
  if (n.indexOf('0') === 0) n = n.slice(1);
  return n.length >= 9 ? n : '';
}
function resolveLogin_(id, password, everyone) {
  id = String(id || '').trim();
  if (id.indexOf('@') !== -1) return id.toLowerCase();
  const pk = phoneKey_(id);
  if (pk && digits_(id).length === id.replace(/[\s+\-()]/g, '').length) {
    const byPhone = everyone.filter(u => phoneKey_(u.phone) === pk || phoneKey_(u.email) === pk);
    const ok = byPhone.filter(u => u.pass === password).map(u => u.email).filter((e, i, a) => a.indexOf(e) === i);
    if (ok.length > 1) throw new Error('الرقم ده متسجل لأكتر من حساب - ادخل بالإيميل');
    if (ok.length) return ok[0];
    if (byPhone.length) return byPhone[0].email;
  }
  const n = normName_(id);
  const byName = everyone.filter(u => u.name && normName_(u.name) === n);
  if (!byName.length) return id.toLowerCase();
  const emails = byName.filter(u => u.pass === password).map(u => u.email).filter((e, i, a) => a.indexOf(e) === i);
  if (emails.length > 1) throw new Error('الاسم ده متكرر عند أكتر من موظف - ادخل بالإيميل');
  return emails.length ? emails[0] : byName[0].email;
}

// أي حد داخل التطبيق (اسمه بيتسجل في تاب اللوجو): d = {pos, size, opacity, logo: "data:image/png;base64,..." أو "" (يشيل اللوجو) أو من غيرها (يسيبه زي ما هو)}
function iiWmSet_(d) {
  const u = iiUser_(d);
  const cur = iiWmRead_();
  const pos = II_WM_POS.indexOf(String(d.pos)) >= 0 ? String(d.pos) : cur.pos;
  const size = Math.max(5, Math.min(100, Math.round(Number(d.size) || cur.size)));
  const op = Math.max(5, Math.min(100, Math.round(Number(d.opacity) || cur.opacity)));
  let logo = cur.logo;
  if (d.logo !== undefined && d.logo !== null) {
    logo = String(d.logo);
    if (logo && !/^data:image\/(png|jpeg|webp);base64,[A-Za-z0-9+\/=]+$/.test(logo)) throw new Error('اللوجو مش واضح - اختار صورة تانية');
    if (logo.length > 400000) throw new Error('اللوجو كبير قوي - اختار صورة أصغر');
  }
  const chunks = [];
  for (let i = 0; i < logo.length; i += 45000) chunks.push(logo.slice(i, i + 45000));
  const ver = String(Date.now());
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = iiWmSheet_();
    if (sh.getLastColumn() > 5 && sh.getLastRow() > 1) sh.getRange(2, 6, 1, sh.getLastColumn() - 5).clearContent();
    sh.getRange(2, 1, 1, 5).setValues([[pos, size, op, ver, u.name]]);
    sh.getRange(2, 4).setNumberFormat('@').setValue(ver);
    if (chunks.length) sh.getRange(2, 6, 1, chunks.length).setNumberFormat('@').setValues([chunks]);
  } finally { lock.releaseLock(); }
  return { wm: { pos: pos, size: size, opacity: op, ver: ver, logo: logo } };
}

// {رقم الصنف: {img: الصورة اللي عليها لوجو, ver: نسخة اللوجو, orig: ملف النسخة الأصلية في Drive}}
function iiMarks_(branch) {
  const out = {};
  try {
    const sh = iiMarkSheet_(), n = sh.getLastRow() - 1;
    if (n < 1) return out;
    const b = posNormBranch_(branch);
    sh.getRange(2, 1, n, 6).getValues().forEach(r => {
      if (posNormBranch_(r[0]) === b) out[String(r[1])] = { img: String(r[2]), ver: String(r[4] || ''), orig: String(r[5] || '') };
    });
  } catch (e) {}
  return out;
}
function iiMark_(branch, id, img, ver, orig) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = iiMarkSheet_(), n = sh.getLastRow() - 1, b = posNormBranch_(branch);
    if (String(sh.getRange(1, 5).getValue()) !== 'نسخة اللوجو') sh.getRange(1, 5, 1, 2).setValues([['نسخة اللوجو', 'ملف الأصل (Drive)']]).setFontWeight('bold');
    if (n > 0) {
      const rows = sh.getRange(2, 1, n, 6).getValues();
      for (let i = 0; i < rows.length; i++) if (posNormBranch_(rows[i][0]) === b && String(rows[i][1]) === String(id)) {
        const old = String(rows[i][5] || '');
        if (old && orig && old !== orig) { try { DriveApp.getFileById(old).setTrashed(true); } catch (e) {} } // أصل أقدم ما بقاش له لازمة
        sh.getRange(i + 2, 3, 1, 4).setNumberFormat('@').setValues([[img, '', String(ver || ''), orig || '']]);
        sh.getRange(i + 2, 4).setNumberFormat('yyyy-mm-dd hh:mm').setValue(new Date());
        return;
      }
    }
    sh.appendRow([branch, String(id), img, new Date(), String(ver || ''), orig || '']);
  } finally { lock.releaseLock(); }
}
function iiUnmark_(branch, id) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = iiMarkSheet_(), n = sh.getLastRow() - 1, b = posNormBranch_(branch);
    if (n < 1) return;
    const rows = sh.getRange(2, 1, n, 2).getValues();
    for (let i = rows.length - 1; i >= 0; i--) if (posNormBranch_(rows[i][0]) === b && String(rows[i][1]) === String(id)) sh.deleteRow(i + 2);
  } finally { lock.releaseLock(); }
}

// ---------- النسخ الأصلية (من غير لوجو) في Drive ----------
// فولدر واحد "مدقق - أصول صور الأصناف"، اسم الملف: الفرع__رقم الصنف__الوقت
const II_ORIG_PROP = 'II_ORIG_FOLDER';
// شغّلها مرة من المحرر: بتعمل الفولدر وبتطلب صلاحية Drive
function setupOrigFolder() {
  Logger.log('فولدر النسخ الأصلية: ' + iiOrigFolder_().getName() + ' ✓');
}
function iiOrigFolder_() {
  const pr = PropertiesService.getScriptProperties();
  const id = pr.getProperty(II_ORIG_PROP);
  if (id) { try { return DriveApp.getFolderById(id); } catch (e) {} }
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const again = pr.getProperty(II_ORIG_PROP);
    if (again && again !== id) return DriveApp.getFolderById(again);
    const f = DriveApp.createFolder('مدقق - أصول صور الأصناف (قبل اللوجو)');
    pr.setProperty(II_ORIG_PROP, f.getId());
    return f;
  } finally { lock.releaseLock(); }
}
function iiOrigName_(branch, id) { return iiTabName_(branch) + '__' + id + '__'; }
function iiOrigSave_(branch, id, blob) {
  const type = String(blob.getContentType() || '');
  const ext = /png/.test(type) ? 'png' : /webp/.test(type) ? 'webp' : /gif/.test(type) ? 'gif' : 'jpg';
  return iiOrigFolder_().createFile(blob.setName(iiOrigName_(branch, id) + Date.now() + '.' + ext)).getId();
}
// الملف ده فعلاً أصل الصنف ده؟ (علشان محدش يبعت رقم ملف تاني)
function iiOrigOk_(branch, id, fileId) {
  try { return DriveApp.getFileById(fileId).getName().indexOf(iiOrigName_(branch, id)) === 0; } catch (e) { return false; }
}
function iiDataUrl_(blob, path) {
  let type = String(blob.getContentType() || '').split(';')[0];
  if (!/^image\//.test(type)) type = /\.png$/i.test(path) ? 'image/png' : /\.webp$/i.test(path) ? 'image/webp' : /\.gif$/i.test(path) ? 'image/gif' : 'image/jpeg';
  return 'data:' + type + ';base64,' + Utilities.base64Encode(blob.getBytes());
}

// الصورة اللي الموبايل هيحط عليها اللوجو - d = {branch, id, bulk} ← {image: "data:...", orig: ملف الأصل}
// لو الصورة عليها لوجو قبل كده وليها نسخة أصلية: بنرجّع الأصلية (علشان اللوجو الجديد ما يتحطش فوق القديم)
// غير كده بنجيب الصورة من البرنامج ونحفظها كنسخة أصلية في Drive.
function iiGet_(d) {
  const b = iiBranch_(d).branch;
  const id = String(d.id || '').replace(/\D/g, '');
  const t = iiRead_(b), i = t.byId[id];
  const path = i === undefined ? '' : t.rows[i][4];
  if (!path || path === II_NONE) throw new Error('الصنف ده مالوش صورة على البرنامج');
  // التطبيق الجماعي (bulk) بيقف عند حد التجهيز علشان يسيب رصيد للشغل العادي
  if (!iiRoom_(8, !d.bulk)) throw new Error(d.bulk ? 'التطبيق الجماعي وقف النهارده علشان رصيد السيرفر - كمّله بكرة من نفس الزرار' : II_QUOTA_MSG);
  const mk = iiMarks_(b)[id];
  if (mk && mk.img === path) {
    if (!mk.orig || !iiOrigOk_(b, id, mk.orig)) throw new Error('الصورة دي عليها لوجو قديم ومالهاش نسخة أصلية محفوظة - صوّر الصنف من جديد');
    return { image: iiDataUrl_(DriveApp.getFileById(mk.orig).getBlob(), path), img: path, orig: mk.orig };
  }
  const cfg = iiCfg_(b);
  if (!cfg) throw new Error('الفرع مش مربوط ببرنامج البيع');
  const res = iiUF_(cfg.url + path, { muteHttpExceptions: true });
  const blob = res.getBlob(), bytes = blob.getBytes();
  if (res.getResponseCode() !== 200 || bytes.length < 100) throw new Error('مقدرتش أجيب الصورة من البرنامج');
  if (bytes.length > 6 * 1024 * 1024) throw new Error('الصورة اللي على البرنامج كبيرة قوي');
  let orig = '';
  try { orig = iiOrigSave_(b, id, blob); } catch (e) { throw new Error('مقدرتش أحفظ النسخة الأصلية في Drive - كلم الدعم (صلاحية Drive)'); }
  return { image: iiDataUrl_(blob, path), img: path, orig: orig };
}
// شيل اللوجو من صورة: بنرجّع النسخة الأصلية على البرنامج - d = {branch, id, bulk}
function iiUnWm_(d) {
  const ub = iiBranch_(d), b = ub.branch;
  const id = String(d.id || '').replace(/\D/g, '');
  const t = iiRead_(b), i = t.byId[id];
  const path = i === undefined ? '' : t.rows[i][4];
  const mk = iiMarks_(b)[id];
  if (!mk || mk.img !== path) throw new Error('الصورة دي مش متسجل إن عليها لوجو');
  if (!mk.orig || !iiOrigOk_(b, id, mk.orig)) throw new Error('الصورة دي مالهاش نسخة أصلية محفوظة - صوّر الصنف من جديد');
  if (!iiRoom_(8, !d.bulk)) throw new Error(d.bulk ? 'التطبيق الجماعي وقف النهارده علشان رصيد السيرفر - كمّله بكرة من نفس الزرار' : II_QUOTA_MSG);
  const blob = DriveApp.getFileById(mk.orig).getBlob();
  let type = String(blob.getContentType() || '');
  if (!/^image\/(jpeg|png|webp)$/.test(type)) type = 'image/jpeg';
  const r = iiSetCore_(b, ub.u.name + ' (شيل اللوجو)', id, Utilities.newBlob(blob.getBytes(), type), null);
  try { iiUnmark_(b, id); } catch (e) {}
  r.wm = ''; r.orig = 0;
  return r;
}
