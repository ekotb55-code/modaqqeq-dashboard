function esc(s) { return String(s ?? '').replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c])); }

function norm(s) {

}
function arErr(e) {
  const m = (e && e.message) ? e.message : String(e || '');
  if (/Failed to fetch|NetworkError|Load failed|network/i.test(m)) return 'مفيش نت أو السيرفر مش بيرد - جرّب تاني';
  if (/JSON|Unexpected token/i.test(m)) return 'السيرفر رد رد غريب - جرّب كمان شوية';
  return /[a-z]{4,}/i.test(m) && !/[؀-ۿ]/.test(m) ? 'حصلت مشكلة - جرّب تاني' : m;
}
async function api(action, data) {
  const body = Object.assign({ email: ls('ii_email') || '', password: ls('ii_pass') || '' }, data || {});
  const res = await fetch(API + '?action=' + action, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify(body) });
  const j = await res.json();
  if (!j.ok) {
    const err = new Error(j.error || 'حصلت مشكلة');
    if (/كلمة السر غلط|موقوف|مالوش صلاحية على تطبيق الصور/.test(j.error || '')) err.auth = true;
    throw err;
  }
  return j;
}

async function openBranch(b) {
  st.branch = b; ls('ii_branch', b);
  $('brName').textContent = b;
  st.items = []; st.byId = {}; st.peekQ = []; st.peekOff = false; st.onlyIds = null; $('q').value = ''; st.q = '';

  let cached = false;
  try {
    const c = await cacheGet(b);
    if (c && st.branch === b) { applyItems(c); st.hash = c.hash || ''; cached = true; }
  } catch (e) {}
  try {

async function loadItems() {
  const b = st.branch;
  const r = await api('imgItems', { branch: b, have: st.hash || '' });
  if (st.branch !== b) return;
  st.scan = r.scan;
  if (r.same && st.items.length) return; // مفيش جديد - القايمة اللي على الموبايل زي ما هي
  st.hash = r.hash || '';
  const y = window.scrollY;
  applyItems({ items: r.items, base: r.base });
  if (y) window.scrollTo(0, y);
  cacheSave();
}
function applyItems(c) {
  const local = {};
  st.items.forEach(it => { if (it.local) local[it.id] = it.local; });
  st.base = c.base;
  st.items = c.items.map(a => ({ id: a[0], name: a[1], code: a[2], cat: a[3] || 'من غير تصنيف', cat0: a[3] || '', img: a[4], wmv: a[5] ? String(a[5]) : '', orig: a[6] ? 1 : 0, key: norm(a[1]), local: local[a[0]] }));
  st.byId = {};
  st.items.forEach(it => { st.byId[it.id] = it; });
  render();
}

// ---------- حفظ القايمة على الموبايل (IndexedDB - مساحة كبيرة ومش بتتمسح زي localStorage) ----------
function idb() {
  if (!idb.p) idb.p = new Promise((ok, no) => {
    const q = indexedDB.open('modaqqeq-images', 1);
    q.onupgradeneeded = () => q.result.createObjectStore('items');
    q.onsuccess = () => ok(q.result);
    q.onerror = () => no(q.error);
  });
  return idb.p;
}
async function cacheGet(b) {
  const db = await idb();
  return new Promise(ok => {
    const q = db.transaction('items').objectStore('items').get(ls('ii_email') + '|' + b);
    q.onsuccess = () => ok(q.result || null); q.onerror = () => ok(null);
  });
}
// بنحفظ بعد كل تغيير (بعد ثانيتين من آخر تغيير علشان ما نحفظش كل شوية)
function cacheSave() {
  clearTimeout(cacheSave.t);
  const b = st.branch;
  cacheSave.t = setTimeout(async () => {
    if (st.branch !== b || !st.items.length) return;

    } catch (e) {}
  }, 2000);
}
async function cacheClear() {

}

async function runScan(restart, quiet) {
  if (st.scanning) return;
  st.scanning = true;
  st.scanP = st.scan && !st.scan.finished ? st.scan : null;
  showProgress();
  const b = st.branch;
  let first = true, lastLoad = Date.now(), errors = 0;
  try {
    for (;;) {
      if (st.branch !== b) break;
      let p;
      try {
        p = await api('imgScan', { branch: b, restart: restart && first });
        errors = 0;
      } catch (err) {
        if (err.auth || ++errors >= 4) throw err;
        await new Promise(r => setTimeout(r, 4000));
        continue;
      }
      first = false;
      st.scan = p; st.scanP = p;
      showProgress();
      if (p.paused) { await loadItems(); if (!quiet) toast('تجهيز الصور وقف النهارده علشان رصيد السيرفر - بيكمل بكرة لوحده، والأصناف اللي قدامك صورها بتظهر عادي'); quiet = true; break; }
      // القايمة بتظهر وهي بتتجهز (كل شوية)
      if (p.finished || Date.now() - lastLoad > 45000 || (!st.items.length && p.total)) { await loadItems(); lastLoad = Date.now(); }
      if (p.finished) break;
    }
    if (!quiet && st.branch === b) toast('الأصناف جاهزة ✓', 'ok');
  } catch (err) {
    if (!quiet) toast(arErr(err), 'bad');
  } finally {
    st.scanning = false;
    st.scanP = null;
    showProgress();
  }
}

function scanLine(p) {
  if (!p || p.finished) return null;
  if (p.paused) return { t: 'تجهيز الصور وقف النهارده (رصيد السيرفر) - بيكمل بكرة • ' + (p.checked || 0).toLocaleString('en-US') + ' من ' + (p.total || 0).toLocaleString('en-US'),
    pct: p.total ? Math.round(100 * p.checked / p.total) : 0 };
  if (p.stage === 'list' || !p.listed) {
    const est = p.pages ? p.pages * 30 : 0;
    return { t: 'بنجيب أصناف الفرع من البرنامج: ' + (p.total || 0).toLocaleString('en-US') + (est ? ' من حوالي ' + est.toLocaleString('en-US') : ''),
      pct: p.pages ? Math.min(100, Math.round(100 * (p.page || 0) / p.pages)) : null };
  }
  return { t: 'بنجيب صور الأصناف: ' + (p.checked || 0).toLocaleString('en-US') + ' من ' + (p.total || 0).toLocaleString('en-US') + ' - تقدر تشتغل عادي',
    pct: p.total ? Math.round(100 * p.checked / p.total) : 0 };
}

function more() {
  const v = st.view, list = $('list');
  if (st.shown >= v.length) return;
  const end = Math.min(v.length, st.shown + PAGE);
  let html = '', lastCat = st.shown ? v[st.shown - 1].cat : null;
  for (let i = st.shown; i < end; i++) {
    const it = v[i];
    if (st.sort === 'cat' && it.cat !== lastCat) { html += '<div class="cat">' + esc(it.cat) + '</div>'; lastCat = it.cat; }

async function pumpPeek() {

async function saveImage(id, getData, wm, bulk) {
  const it = st.byId[id];
  let res = { ok: false, err: '' };
  const sv = st.saves[id] = { name: it.name, ph: 'prep', pct: 5, t0: Date.now() };

  const tick = setInterval(() => {
    if (sv.ph !== 'up') return;
    const t = (Date.now() - sv.t1) / 1000;
    sv.pct = Math.min(95, Math.round(25 + 70 * (1 - Math.exp(-t / 3.5))));
    showProgress(); setRowPct(id, sv.pct);
  }, 300);
  try {
    let got = await getData();
    if (typeof got === 'string') got = { data: got };
    const data = got.data || '';
    if (data) sv.preview = data;
    sv.ph = 'up'; sv.pct = 25; sv.t1 = Date.now();
    refreshRow(id); showProgress();
    const r = got.done || await api('imgSet', { branch: st.branch, id: id, image: data, wm: wm ? 1 : 0, wmVer: wm && st.wm ? st.wm.ver : '', clean: got.clean || '', orig: got.orig || '' });
    it.img = r.img;
    it.wmv = r.wm ? String(r.wm) : ''; it.orig = r.orig ? 1 : 0;
    it.local = !bulk && data ? data : undefined; // الصورة اللي اتصورت تفضل ظاهرة على طول (من غير ما نستنى البرنامج)
    res.ok = true;
    st.hash = ''; cacheSave();
    sv.ph = 'done'; sv.pct = 100;
    delete st.busy[id];
    refreshRow(id);
    const ph = document.querySelector('#it' + id + ' .ph');
    if (ph) ph.classList.add('done');
    if (!bulk) toast('اتحفظت الصورة على البرنامج ✓', 'ok');
    $('nNone').textContent = '(' + st.items.filter(x => x.img === '-').length + ')';
  } catch (err) {
    sv.ph = 'err'; sv.pct = 100;
    delete st.busy[id];
    refreshRow(id);
    res.err = arErr(err); res.auth = !!err.auth;
    if (err.auth) { toast(arErr(err), 'bad'); clearInterval(tick); delete st.saves[id]; showProgress(); showLogin(); return res; }
    if (!bulk) toast(arErr(err), 'bad');
  } finally {
    clearInterval(tick);
    showProgress();
    // الحالة الأخيرة بتفضل ظاهرة شوية وبعدين تختفي
    setTimeout(() => { if (st.saves[id] === sv) { delete st.saves[id]; showProgress(); } }, bulk ? 600 : sv.ph === 'err' ? 6000 : 2500);
  }
  return res;
}

async function squareJpeg(file, size, q) {
  let src;
  try { src = await createImageBitmap(file); } catch (e) {
    src = await new Promise((ok, bad) => {
      const im = new Image(); im.onload = () => ok(im); im.onerror = () => bad(new Error('الصورة دي مش بتتفتح - اختار صورة تانية'));
      im.src = URL.createObjectURL(file);
    });
  }
  const w = src.width, h = src.height, s = Math.min(w, h);
  if (!s) throw new Error('الصورة دي مش بتتفتح - اختار صورة تانية');
  const out = Math.min(size, s);
  const c = document.createElement('canvas');
  c.width = out; c.height = out;
  const g = c.getContext('2d');
  g.fillStyle = '#fff'; g.fillRect(0, 0, out, out);
  g.imageSmoothingQuality = 'high';
  g.drawImage(src, (w - s) / 2, (h - s) / 2, s, s, 0, 0, out, out);
  return c.toDataURL('image/jpeg', q);
}

// مكان اللوجو على الصورة: tr فوق يمين، tl فوق شمال، br تحت يمين، bl تحت شمال، c في النص
function wmBox(W, H, w, im) {
  let lw = W * w.size / 100, lh = lw * im.height / im.width;
  if (lh > H * 0.92) { lh = H * 0.92; lw = lh * im.width / im.height; }
  const m = Math.round(W * 0.03);
  if (w.pos === 'c') return { x: (W - lw) / 2, y: (H - lh) / 2, w: lw, h: lh };
  return { x: w.pos[1] === 'r' ? W - lw - m : m, y: w.pos[0] === 't' ? m : H - lh - m, w: lw, h: lh };
}
function stamp(c) {
  if (!wmReady()) return;
  const g = c.getContext('2d'), b = wmBox(c.width, c.height, st.wm, st.wmImg);
  g.save(); g.globalAlpha = st.wm.opacity / 100; g.imageSmoothingQuality = 'high';
  g.drawImage(st.wmImg, b.x, b.y, b.w, b.h); g.restore();
}
// صورة موجودة (من البرنامج) ← نفس الصورة وعليها اللوجو
async function stampDataUrl(data) {
  const im = await loadImg(data);
  const k = Math.min(1, 1000 / Math.max(im.width, im.height));
  const c = document.createElement('canvas');
  c.width = Math.round(im.width * k); c.height = Math.round(im.height * k);
  if (!c.width || !c.height) throw new Error('الصورة اللي على البرنامج مش بتتفتح');
  const g = c.getContext('2d');
  g.fillStyle = '#fff'; g.fillRect(0, 0, c.width, c.height);
  g.imageSmoothingQuality = 'high';
  g.drawImage(im, 0, 0, c.width, c.height);
  stamp(c);
  return c.toDataURL('image/jpeg', 0.88);
}

async function bulkWm(mode) {
  if (st.bulk) { st.bulk.stop = true; return; }
  const add = mode !== 'off';
  if (add && !wmReady()) return toast('ارفع اللوجو واحفظ الإعدادات الأول', 'bad');
  const todo = (add ? wmTodo() : wmOff()).map(it => it.id), b = st.branch;
  if (!todo.length) return toast(add ? 'كل صور الفرع عليها اللوجو ✓' : 'مفيش صور ينفع يتشال منها اللوجو', 'ok');
  if (!confirm((add ? 'هنحط اللوجو على ' : 'هنشيل اللوجو ونرجّع الصورة الأصلية لـ ') + todo.length + ' صورة في فرع "' + b + '" ونرفعهم تاني على البرنامج.\nسيب التطبيق مفتوح لحد ما يخلص. نبدأ؟')) return;
  const bk = st.bulk = { n: 0, fail: 0, total: todo.length, stop: false, title: add ? 'لوجو على كل الصور' : 'شيل اللوجو من كل الصور' };
  let streak = 0, why = '';
  showProgress();
  for (const id of todo) {
    if (bk.stop || st.branch !== b) break;
    const it = st.byId[id];
    if (!it || st.busy[id] || !(add ? wmCan(it) : (it.wmv && it.orig))) continue;
    const r = await (add ? addLogo(id, true) : unLogo(id, true));
    if (r.ok) { bk.n++; streak = 0; }
    else {
      if (r.auth) break;
      if (/رصيد السيرفر/.test(r.err)) { why = r.err; break; }
      bk.fail++;
      if (++streak >= 5) { why = 'وقفنا: 5 صور ورا بعض ما اتحفظتش - ' + r.err; break; }
    }
    showProgress();
    await new Promise(z => setTimeout(z, 1500)); // استراحة بين كل صورة والتانية
  }
  st.bulk = null;
  showProgress();
  toast(why || ((add ? 'اللوجو اتحط على ' : 'اللوجو اتشال من ') + bk.n + ' صورة' + (bk.fail ? ' • ' + bk.fail + ' ما اتحفظتش' : '') + (bk.stop ? ' (وقّفته)' : ' ✓')), why ? 'bad' : 'ok');
}

async function findCode(code) {
  code = String(code || '').trim();
  if (!code) return;
  $('q').value = code; st.q = code; st.onlyIds = null;
  const local = st.items.filter(it => it.code === code);
  if (local.length) { st.q = ''; st.onlyIds = local.map(x => x.id); render(); return; }
  toast('بندوّر على الباركود في البرنامج...');
  try {
    const r = await api('imgFind', { branch: st.branch, code: code });
    if (!r.ids.length) { st.onlyIds = []; render(); return toast('الباركود ده مش موجود في البرنامج', 'bad'); }
    // صنف جديد مش في القايمة لسه
    (r.items || []).forEach(a => {
      if (!st.byId[a[0]]) { const it = { id: a[0], name: a[1], code: a[2], cat: a[3] || 'من غير تصنيف', img: '', key: norm(a[1]) }; st.items.push(it); st.byId[it.id] = it; }
    });
    st.q = ''; st.onlyIds = r.ids; render();
  } catch (err) { toast(arErr(err), 'bad'); }
}
let zx = null, stream = null, stopScan = null;
function loadZx() {
  if (window.ZXing) return Promise.resolve();
  return new Promise((ok, bad) => {
    const s = document.createElement('script');
    s.src = 'https://cdn.jsdelivr.net/npm/@zxing/library@0.21.3/umd/index.min.js';
    s.onload = ok; s.onerror = () => bad(new Error('مقدرتش أشغّل قارئ الباركود - اتأكد من النت'));

$('btnScan').onclick = async () => {
  $('scan').classList.remove('hidden');
  const vid = $('vid');
  let done = false;
  const finish = code => {
    if (done) return; done = true;
    closeScan();
    if (code) { if (navigator.vibrate) navigator.vibrate(60); findCode(code); }
  };
  try {
    stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } }, audio: false });
    vid.srcObject = stream;
    await vid.play();
    if ('BarcodeDetector' in window) {
      const det = new BarcodeDetector({ formats: ['ean_13', 'ean_8', 'code_128', 'code_39', 'upc_a', 'upc_e', 'itf', 'qr_code'] });
      let alive = true;
      stopScan = () => { alive = false; };
      const tick = async () => {
        if (!alive || done) return;
        try { const r = await det.detect(vid); if (r && r[0] && r[0].rawValue) return finish(r[0].rawValue); } catch (e) {}
        setTimeout(tick, 180);
      };
      tick();
    } else {
      await loadZx();
      if (done) return;
      zx = new ZXing.BrowserMultiFormatReader();
      const cb = res => { if (res) finish(res.getText()); };
      if (typeof zx.decodeFromStream === 'function') zx.decodeFromStream(stream, vid, cb);
      else zx.decodeFromVideoElementContinuously(vid, cb);
      stopScan = () => { try { zx.reset(); } catch (e) {} };
    }
  } catch (err) {
    closeScan();
    toast(/Permission|NotAllowed|denied/i.test(String(err && (err.name || err.message))) ? 'لازم تسمح للتطبيق يستخدم الكاميرا' : arErr(err), 'bad');
  }
};

async function shareApp() {
  const url = location.origin + '/modaqqeq-images/';
  const text = 'تطبيق صور الأصناف - مدقق\nافتح الرابط وادخل بنفس اسمك وكلمة السر، وتقدر تثبّته على الموبايل من زرار «تثبيت التطبيق»:';
  if (navigator.share) {
    try { await navigator.share({ title: 'تطبيق صور الأصناف - مدقق', text: text, url: url }); return; }
    catch (e) { if (e && e.name === 'AbortError') return; }
  }
  try { await navigator.clipboard.writeText(url); toast('اتنسخ رابط التطبيق ✓', 'ok'); } catch (e) {}
  window.open('https://wa.me/?text=' + encodeURIComponent(text + '\n' + url), '_blank');
}
$('shareBtn').onclick = shareApp;
$('shareBtn2').onclick = shareApp;
showInstall();
