// الدخول: نفس بيانات تاب الموظفين + لازم الاسم موجود في تاب "تطبيق الصور"
function iiUser_(d) {
  const pass = String(d.password || '').trim();
  if (!String(d.email || '').trim() || !pass) throw new Error('اكتب اسمك (أو الإيميل) وكلمة السر');
  const everyone = users_();
  const email = resolveLogin_(d.email, pass, everyone);
  const me = everyone.filter(u => u.email === email && u.pass === pass);
  if (!me.length) throw new Error('الاسم (أو الإيميل) أو كلمة السر غلط');
  if (!me.some(u => u.active)) throw new Error('الحساب ده موقوف - كلم المدير');
  const keys = [normName_(email)].concat(me.map(u => normName_(u.name)).filter(Boolean));
  const sh = iiAccessSheet_();
  const rows = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, 3).getValues() : [];
  const linked = iiLinked_();
  let all = false;
  const want = [];
  rows.forEach(r => {
    if (keys.indexOf(normName_(r[0])) < 0) return;
    if (r[2] === false || String(r[2]).toUpperCase() === 'FALSE') return;
    splitStores_(r[1]).forEach(b => { if (normName_(b) === normName_(ALL)) all = true; else want.push(b); });
  });
  if (!all && !want.length) throw new Error('حسابك مالوش صلاحية على تطبيق الصور - كلم المدير');
  const branches = all ? linked : linked.filter(b => want.some(w => posNormBranch_(w) === posNormBranch_(b)));
  if (!branches.length) throw new Error('الفرع بتاعك مش مربوط ببرنامج البيع - كلم الدعم');
  return { name: me[0].name || email, email: email, branches: branches, all: all };
}
function iiBranch_(d) {
  const u = iiUser_(d);
  const b = u.branches.find(x => posNormBranch_(x) === posNormBranch_(d.branch));
  if (!b) throw new Error('الفرع ده مش من صلاحياتك');
  return { u: u, branch: b };
}

function iiLogin_(d) {
  const u = iiUser_(d);
  return { name: u.name, email: u.email, branches: u.branches };
}

function iiTabName_(branch) { return String(branch).replace(/[\[\]*?:\\\/']/g, ' ').trim().slice(0, 90); }
function iiSheet_(branch) {
  const ss = iiCacheSS_();
  let sh = ss.getSheetByName(iiTabName_(branch));
  if (!sh) {
    sh = ss.insertSheet(iiTabName_(branch));
    sh.getRange(1, 1, 1, II_COLS.length).setValues([II_COLS]).setFontWeight('bold');
    sh.setFrozenRows(1);
    sh.setRightToLeft(true);
    sh.getRange('A:C').setNumberFormat('@');

// {sh, rows: [[id,name,code,cat,img,seen]], byId: {id: index}}
function iiRead_(branch) {
  const sh = iiSheet_(branch), n = sh.getLastRow() - 1;
  const rows = n > 0 ? sh.getRange(2, 1, n, II_COLS.length).getValues().map(r => r.map(x => String(x == null ? '' : x))) : [];
  const byId = {}, origE = {};
  rows.forEach((r, i) => { if (r[0]) { byId[r[0]] = i; origE[r[0]] = r[4]; } });
  return { sh: sh, rows: rows, byId: byId, origE: origE };
}
// بنكتب الجدول كله مرة واحدة - والصورة اللي حد حفظها في نفس الوقت ما تضيعش
function iiWrite_(t) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const n = t.sh.getLastRow() - 1;
    if (n > 0) {
      t.sh.getRange(2, 1, n, 5).getValues().forEach(r => {
        const id = String(r[0]), e = String(r[4] == null ? '' : r[4]), i = t.byId[id];
        if (i !== undefined && e !== t.origE[id] && t.rows[i] && t.rows[i][0] === id) t.rows[i][4] = e;
      });
    }
    if (t.full && n > t.rows.length) t.sh.getRange(t.rows.length + 2, 1, n - t.rows.length, II_COLS.length).clearContent();
    if (t.rows.length) t.sh.getRange(2, 1, t.rows.length, II_COLS.length).setValues(t.rows);
    t.byId = {}; t.origE = {};
    t.rows.forEach((r, i) => { t.byId[r[0]] = i; t.origE[r[0]] = r[4]; });
  } finally { lock.releaseLock(); }
}
// صورة صنف واحد في الشيت (بعد الحفظ)
function iiSetCell_(branch, id, img) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sh = iiSheet_(branch), n = sh.getLastRow() - 1;
    if (n < 1) return;
    const ids = sh.getRange(2, 1, n, 1).getValues();
    for (let i = 0; i < ids.length; i++) if (String(ids[i][0]) === String(id)) { sh.getRange(i + 2, 5).setValue(img); return; }
  } finally { lock.releaseLock(); }
}

// ---------- الدخول على برنامج الفرع (الجلسة بتتحفظ 20 دقيقة) ----------
function iiKey_(branch) { return 'iijar_' + Utilities.base64EncodeWebSafe(Utilities.newBlob(posNormBranch_(branch)).getBytes()).slice(0, 180); }
function iiSession_(branch, fresh) {
  const c = CacheService.getScriptCache(), k = iiKey_(branch);
  if (!fresh) {
    try {
      const jar = JSON.parse(c.get(k) || 'null');
      const cfg = posCfgFromSheet_(branch);
      if (jar && cfg) return { cfg: cfg, jar: jar, branch: branch };
    } catch (e) {}
  }
  const s = bsLogin_(branch);
  s.branch = branch;
  c.put(k, JSON.stringify(s.jar), 1200);
  return s;
}
function iiExpired_(res) {
  const code = res.getResponseCode();
  if (code >= 300 && code < 400) return true;
  const t = res.getContentText('UTF-8');
  return /loginform|logincheck\.php/i.test(t.slice(0, 60000)) && !/show_items_cat|edit_item_save|preview_item|id=["']?row\d+/.test(t);
}
function iiReq_(s, path, payload) {
  const r = {
    url: s.cfg.url + path, muteHttpExceptions: true, followRedirects: false,
    headers: { Cookie: Object.keys(s.jar).map(k => k + '=' + s.jar[k]).join('; '), 'User-Agent': 'Mozilla/5.0 (Modaqqeq images)' }
  };
  if (payload) { r.method = 'post'; r.payload = payload; }
  return r;
}
// طلبات كتير مع بعض - لو الجلسة خلصت ندخل تاني مرة واحدة
function iiFetchAll_(s, build) {
  let res = UrlFetchApp.fetchAll(build(s));
  if (res.some(iiExpired_)) {
    const s2 = iiSession_(s.branch, true);
    s.jar = s2.jar;
    res = UrlFetchApp.fetchAll(build(s));
    if (res.some(iiExpired_)) throw new Error('برنامج البيع بيخرّج الجلسة - جرّب كمان شوية');
  }
  return res;
}
// طلب واحد (لو الجلسة خلصت ندخل تاني)
function iiFetch_(s, path, opt) {
  let res = bsFetch_(s, path, opt);
  if (iiExpired_(res)) {
    const s2 = iiSession_(s.branch, true);
    s.jar = s2.jar;
    res = bsFetch_(s, path, opt);
  }
  return res;
}
function iiSaveJar_(s) { CacheService.getScriptCache().put(iiKey_(s.branch), JSON.stringify(s.jar), 1200); }

// ---------- قايمة الأصناف ----------
// صف من جدول الأصناف في البرنامج ← [id, name, code, cat]
function iiParseRows_(html) {
  const out = [];
  (String(html).match(/<tr[^>]*id=["']?row\d+[\s\S]*?<\/tr>/gi) || []).forEach(tr => {
    const id = (tr.match(/id=["']?row(\d+)/i) || [])[1];
    if (!id) return;
    const pt = (tr.match(/<a[^>]*class=["'][^"']*print_ticket[^>]*>/i) || [])[0] || '';
    let name = pt ? iiAttr_(pt, 'data-name') : '';
    if (!name) name = iiDecode_(((tr.match(/item\.php\?id=\d+["']\s*>([^<]+)<\/a>/i) || [])[1] || '').trim());
    const code = pt ? (iiAttr_(pt, 'data-barcode') || '') : '';
    const cat = iiDecode_(((tr.match(/show_items_cat\(\d+\)["'][^>]*>([^<]*)</i) || [])[1] || '').trim());
    out.push([id, String(name || '').trim(), String(code).trim(), cat]);
  });
  return out;
}
function iiPagesOf_(html) {
  let mx = 1;
  (String(html).match(/current\s*=\s*(\d+)/g) || []).forEach(x => { const n = Number(x.replace(/\D/g, '')); if (n > mx) mx = n; });
  (String(html).match(/search_items\((\d+)\)/g) || []).forEach(x => { const n = Number(x.replace(/\D/g, '')); if (n > mx) mx = n; });
  return mx;
}
function iiListPayload_(p) {
  return { current: String(p), searchval: '', store: '', select_cat: '', comp_id: '', barcode_only: '', factory_id: '' };
}
// رابط الصورة من صفحة الصنف (item.php) ← "upload/..." أو "-"
function iiImgOf_(html) {
  const m = String(html).match(/<img\b[^>]*class=["'][^"']*preview_item[^"']*["'][^>]*>/i) || String(html).match(/<img\b[^>]*preview_item[^>]*>/i);
  if (m) {
    const src = iiAttr_(m[0], 'src') || '';
    const u = src.match(II_IMG_RE);
    return u ? u[0] : II_NONE;
  }
  return II_NONE;
}

function iiScanKey_(branch) { return 'II_SCAN_' + Utilities.base64EncodeWebSafe(Utilities.newBlob(posNormBranch_(branch)).getBytes()).slice(0, 120); }
function iiScanState_(branch) {
  try { return JSON.parse(PropertiesService.getScriptProperties().getProperty(iiScanKey_(branch)) || 'null'); } catch (e) { return null; }
}
function iiScanSave_(branch, st) { PropertiesService.getScriptProperties().setProperty(iiScanKey_(branch), JSON.stringify(st)); }

function iiProgress_(t, st) {
  return {
    stage: st ? st.stage : 'none', finished: !!(st && st.stage === 'done'), listed: !!(st && st.fullSeen),
    page: st ? Math.min(st.page || 1, st.pages || st.page || 1) : 0, pages: st ? st.pages || 0 : 0,
    total: t.rows.length, checked: t.rows.filter(r => r[4]).length,
    withImg: t.rows.filter(r => r[4] && r[4] !== II_NONE).length, done: st && st.done || ''
  };
}

// قفل لكل فرع لوحده (علشان تجهيز فرع ما يوقفش حفظ صورة في فرع تاني)
function iiMutex_(key, ttl, waitMs) {
  const c = CacheService.getScriptCache(), end = Date.now() + (waitMs || 0);
  for (;;) {
    const l = LockService.getScriptLock();
    if (l.tryLock(10000)) {
      try {
        if (!c.get(key)) { c.put(key, '1', ttl); return true; }
      } finally { l.releaseLock(); }
    }
    if (Date.now() > end) return false;
    Utilities.sleep(1500);
  }
}
function iiUnmutex_(key) { CacheService.getScriptCache().remove(key); }
function iiMKey_(kind, branch) { return 'iim_' + kind + '_' + Utilities.base64EncodeWebSafe(Utilities.newBlob(posNormBranch_(branch)).getBytes()).slice(0, 150); }

// خطوة تجهيز (حوالي 40 ثانية) - التطبيق بينادي لحد ما finished
// المرحلة 1: صفحات قايمة الأصناف (30 صنف في الصفحة) ← رقم كل صنف واسمه وباركوده وتصنيفه
// المرحلة 2: صفحة كل صنف لسه ما اتعرفتش صورته ← رابط الصورة
// d.restart = يبدأ من الأول (أصناف جديدة / أسماء اتغيرت)
function iiScan_(d) {
  return iiScanCore_(iiBranch_(d).branch, !!d.restart);
}
function iiScanCore_(b, restart) {
  const d = { restart: restart };
  const mk = iiMKey_('scan', b);
  if (!iiMutex_(mk, 90, 0)) return iiProgress_(iiRead_(b), iiScanState_(b)); // فيه حد تاني بيجهّز نفس الفرع
  try {
    const t0 = Date.now(), budget = 40 * 1000;
    let st = iiScanState_(b);
    if (!st || (d.restart && st.stage === 'done')) st = { stage: 'list', page: 1, pages: 0, run: String(Date.now()), started: new Date().toISOString(), fullSeen: st && st.fullSeen || '' };
    const t = iiRead_(b);
    const s = iiSession_(b);
    if (st.stage === 'list') {
      while (Date.now() - t0 < budget && (!st.pages || st.page <= st.pages)) {
        const batch = [];
        for (let p = st.page; p < st.page + 8 && (!st.pages || p <= st.pages); p++) batch.push(p);
        if (!batch.length) break;
        const res = iiFetchAll_(s, x => batch.map(p => iiReq_(x, 'items/show_items.php', iiListPayload_(p))));
        res.forEach(r => {
          const html = r.getContentText('UTF-8');
          if (!st.pages) st.pages = iiPagesOf_(html);
          iiParseRows_(html).forEach(it => {
            const i = t.byId[it[0]];
            if (i === undefined) { t.byId[it[0]] = t.rows.length; t.rows.push([it[0], it[1], it[2], it[3], '', st.run]); }
            else { const row = t.rows[i]; row[1] = it[1]; row[2] = it[2]; row[3] = it[3]; row[5] = st.run; }
          });
        });
        st.page = batch[batch.length - 1] + 1;
      }
      if (st.pages && st.page > st.pages) {
        // الأصناف اللي اتمسحت من البرنامج: نشيلها
        t.rows = t.rows.filter(r => r[5] === st.run);
        t.byId = {};
        t.rows.forEach((r, i) => { t.byId[r[0]] = i; });
        t.full = true;
        st.fullSeen = st.run;
        st.stage = 'imgs';
      }
      iiWrite_(t);
    }
    if (st.stage === 'imgs' && Date.now() - t0 < budget) {
      const todo = [];
      t.rows.forEach((r, i) => { if (!r[4]) todo.push(i); });
      let k = 0;
      while (Date.now() - t0 < budget && k < todo.length) {
        const part = todo.slice(k, k + 30);
        const res = iiFetchAll_(s, x => part.map(i => iiReq_(x, 'items/item.php?id=' + t.rows[i][0])));
        part.forEach((i, j) => {
          const html = res[j].getContentText('UTF-8');
          if (res[j].getResponseCode() === 200 && html.length > 1000) t.rows[i][4] = iiImgOf_(html);
        });
        k += part.length;
      }
      iiWrite_(t);
      if (k >= todo.length) { st.stage = 'done'; st.done = new Date().toISOString(); }
    }
    iiSaveJar_(s);
    iiScanSave_(b, st);
    return iiProgress_(t, st);
  } finally {
    iiUnmutex_(mk);
  }
}

// القايمة للتطبيق ← {items: [[id, name, code, cat, img]], base, scan}
function iiItems_(d) {
  const b = iiBranch_(d).branch;
  const cfg = posCfgFromSheet_(b);
  if (!cfg) throw new Error('الفرع مش مربوط ببرنامج البيع');
  const t = iiRead_(b), st = iiScanState_(b);
  const items = t.rows.filter(r => r[0]).map(r => [r[0], r[1], r[2], r[3], r[4]]);
  return { items: items, base: cfg.url, scan: iiProgress_(t, st) };
}

// البحث بالباركود (الأساسي أو الفرعي) ← {ids: [...]}
function iiFind_(d) {
  const b = iiBranch_(d).branch;
  const code = String(d.code || '').trim();
  if (!code) return { ids: [] };
  const s = iiSession_(b);
  const res = iiFetchAll_(s, x => [iiReq_(x, 'items/show_items.php',
    { current: '1', searchval: code, store: '', select_cat: '', comp_id: '', barcode_only: '1', factory_id: '' })]);
  iiSaveJar_(s);
  const rows = iiParseRows_(res[0].getContentText('UTF-8'));
  return { ids: rows.slice(0, 10).map(r => r[0]), items: rows.slice(0, 10) };
}

// ---------- حفظ الصورة ----------
// d = {branch, id, image: "data:image/jpeg;base64,..."}
function iiSet_(d) {
  const ub = iiBranch_(d);
  const m = String(d.image || '').match(/^data:(image\/(?:jpeg|png|webp));base64,(.+)$/);
  if (!m) throw new Error('الصورة مش واضحة - اختار صورة تانية');
  return iiSetCore_(ub.branch, ub.u.name, d.id, Utilities.newBlob(Utilities.base64Decode(m[2]), m[1]), null);
}
// blob = الصورة الجديدة، أو path = رابط صورة موجودة على البرنامج (upload/... - للرجوع بعد التجربة)
function iiSetCore_(b, who, id, blobIn, path) {
  id = String(id || '').replace(/\D/g, '');
  if (!id) throw new Error('الصنف مش معروف');
  const type = blobIn ? blobIn.getContentType() : '';
  const bytes = blobIn ? blobIn.getBytes() : null;
  if (bytes && bytes.length > 3 * 1024 * 1024) throw new Error('الصورة كبيرة قوي');
  const ext = type === 'image/png' ? 'png' : type === 'image/webp' ? 'webp' : 'jpg';
  const mk = iiMKey_('set', b);
  if (!iiMutex_(mk, 120, 45000)) throw new Error('فيه صورة تانية بتتحفظ دلوقتي - جرّب تاني');
  let name = '', oldImg = '';
  try {
    const s = iiSession_(b);
    // 1) صفحة تعديل الصنف (قبل أي حاجة: نتأكد إننا نقدر نقراها كويس)
    const edit = () => iiFetch_(s, 'items/new_item.php', { method: 'post', payload: { id: id, edit: 'item', store_id: '' } }).getContentText('UTF-8');
    let html = edit();
    if (!/edit_item_save\.php/.test(html)) throw new Error('مقدرتش أفتح صفحة تعديل الصنف في البرنامج');
    const before = iiFormPayload_(html, null);
    if (before.id !== id) throw new Error('البرنامج فتح صنف تاني - ما اتحفظش حاجة');
    name = before.payload.item_name || '';
    oldImg = before.oldImg;
    // 2) رفع الصورة
    let newImg = path;
    if (bytes) {
      const blob = Utilities.newBlob(bytes, type, 'item_' + id + '.' + ext);
      const up = iiFetch_(s, 'uploader.php', { method: 'post', ajax: true, payload: { photoimg: blob } });
      const file = String(up.getContentText('UTF-8') || '').trim().replace(/^"+|"+$/g, '');
      if (!/^[\w\-\/.]+\.(?:jpe?g|png|gif|webp)$/i.test(file)) {
        iiLog_(b, who, id, name, oldImg, '', 'رفع الصورة فشل', file.slice(0, 200));
        throw new Error('البرنامج رفض رفع الصورة' + (file && file.length < 120 ? ': ' + file.replace(/<[^>]+>/g, ' ').trim() : ''));
      }
      newImg = 'upload/' + file.replace(/^upload\//, '');
    }
    if (newImg == null) throw new Error('مفيش صورة');
    // 3) الحفظ: نفس بيانات الصفحة بالظبط والصورة الجديدة بس (بنقرا الصفحة تاني علشان تبقى آخر حاجة)
    html = edit();
    const form = iiFormPayload_(html, newImg);
    if (form.id !== id) throw new Error('البرنامج فتح صنف تاني - ما اتحفظش حاجة');
    const sent = iiParam_(form.payload);
    const res = iiFetch_(s, 'items/edit_item_save.php', { method: 'post', ajax: true, form: true, payload: sent });
    const msg = String(res.getContentText('UTF-8') || '').trim();
    if (!(Number(msg) > 0)) {
      iiLog_(b, who, id, name, oldImg, newImg, 'الحفظ فشل', msg.slice(0, 300));
      const txt = msg.replace(/<[^>]+>/g, ' ').replace(/\s+/g, ' ').trim();
      throw new Error('البرنامج ما حفظش الصورة' + (txt && txt.length < 150 ? ': ' + txt : ' - غالباً يوزر الربط مالوش صلاحية تعديل الأصناف'));
    }
    // 4) التأكد: نفتح الصنف تاني ونقارن كل الخانات (ما عدا الكمية لأنها بتتغير مع البيع)
    const after = iiFormPayload_(edit(), null);
    const diffs = [];
    const skip = { quantity: 1, old_quantity: 1 };
    const pa = iiParam_(after.payload).split('&'), pb = sent.split('&');
    const A = {}, B = {};
    pa.forEach(x => { const k = decodeURIComponent(x.split('=')[0]); if (!skip[k]) A[x] = 1; });
    pb.forEach(x => { const k = decodeURIComponent(x.split('=')[0]); if (!skip[k]) B[x] = 1; });
    Object.keys(B).forEach(x => { if (!A[x]) diffs.push(decodeURIComponent(x.split('=')[0])); });
    Object.keys(A).forEach(x => { if (!B[x]) { const k = decodeURIComponent(x.split('=')[0]); if (diffs.indexOf(k) < 0) diffs.push(k); } });
    const ok = after.oldImg === newImg;
    // 5) شيت الصور
    try { iiSetCell_(b, id, II_IMG_RE.test(newImg) ? newImg.match(II_IMG_RE)[0] : II_NONE); } catch (e) {}
    iiSaveJar_(s);
    iiLog_(b, who, id, name, oldImg, newImg, ok ? (diffs.length ? 'اتحفظت - في خانات اتغيرت' : 'تمام') : 'الصورة ما اتغيرتش',
      diffs.length ? 'اتغير: ' + diffs.join('، ') : '');
    if (!ok) throw new Error('البرنامج رد إن الحفظ تمام بس الصورة ما اتغيرتش - كلم الدعم');
    return { img: II_IMG_RE.test(newImg) ? newImg : II_NONE, url: s.cfg.url + newImg, changed: diffs, oldImg: oldImg, name: name };
  } finally {
    iiUnmutex_(mk);
  }
}

// ---------- قراءة صفحة تعديل الصنف (نفس اللي الصفحة بتبعته بالظبط) ----------
function iiDecode_(s) {
  const named = { quot: '"', apos: "'", lt: '<', gt: '>', amp: '&', nbsp: '\u00a0' };
  return String(s == null ? '' : s).replace(/&(#\d+|#x[0-9a-f]+|quot|apos|lt|gt|amp|nbsp|#039);/gi, (m, e) => {
    if (e[0] === '#') return String.fromCharCode(e[1] === 'x' || e[1] === 'X' ? parseInt(e.slice(2), 16) : Number(e.slice(1)));
    return named[e.toLowerCase()];
  });
}
function iiAttr_(tag, name) {
  const m = String(tag).match(new RegExp('\\s' + name + '\\s*=\\s*(?:"([^"]*)"|\'([^\']*)\'|([^\\s>]+))', 'i'));
  return m ? iiDecode_(m[1] != null ? m[1] : (m[2] != null ? m[2] : m[3])) : null;
}
function iiHasAttr_(tag, name) { return new RegExp('\\s' + name + '(?=[\\s=/>])', 'i').test(String(tag)); }
// كل التاجات من نوع معين (من غير اللي جوه السكريبتات)
function iiTags_(html, tagName) {
  return String(html).match(new RegExp('<' + tagName + '\\b[^>]*>', 'gi')) || [];
}
function iiFormPayload_(html, newImg) {
  // نشيل السكريبتات والتعليقات علشان التاجات اللي جواهم ما تتحسبش
  const clean = String(html).replace(/<script\b[\s\S]*?<\/script>/gi, '').replace(/<!--[\s\S]*?-->/g, '');
  const inputs = iiTags_(clean, 'input');
  const byId = id => inputs.find(t => iiAttr_(t, 'id') === id);
  const val = id => { const t = byId(id); return t ? (iiAttr_(t, 'value') || '') : undefined; };
  const checked = id => { const t = byId(id); return !!(t && iiHasAttr_(t, 'checked')); };
  const radio = name => {
    const t = inputs.find(x => iiAttr_(x, 'name') === name && /type=["']?radio/i.test(x) && iiHasAttr_(x, 'checked'));
    return t ? (iiAttr_(t, 'value') || 'on') : undefined;
  };
  const radioClass = cls => {
    const t = inputs.find(x => (' ' + (iiAttr_(x, 'class') || '') + ' ').indexOf(' ' + cls + ' ') >= 0 && iiHasAttr_(x, 'checked'));
    return t ? (iiAttr_(t, 'value') || 'on') : undefined;
  };
  const selects = [];
  clean.replace(/<select\b([^>]*)>([\s\S]*?)<\/select>/gi, (m, a, body) => { selects.push({ tag: '<select' + a + '>', body: body }); return m; });
  const sel = id => {
    const s = selects.find(x => iiAttr_(x.tag, 'id') === id);
    if (!s) return undefined;
    const opts = s.body.match(/<option\b[^>]*>/gi) || [];
    const multi = iiHasAttr_(s.tag, 'multiple');
    const on = opts.filter(o => iiHasAttr_(o, 'selected')).map(o => { const v = iiAttr_(o, 'value'); return v == null ? '' : v; });
    if (multi) return on.length ? on : null;
    if (on.length) return on[on.length - 1];
    if (!opts.length) return null;
    const v = iiAttr_(opts[0], 'value');
    return v == null ? '' : v;
  };
  const ta = (clean.match(/<textarea\b[^>]*id=["']details["'][^>]*>([\s\S]*?)<\/textarea>/i) || [])[1];
  const img = inputs.length >= 0 ? (String(html).match(/<img\b[^>]*id=["']upload_img["'][^>]*>/i) || [])[0] : '';
  const oldImg = img ? (iiAttr_(img, 'newlink') || '') : '';
  // آخر جزء في datasend: "multi_price":"1","id":"..","old_quantity":".."
  const block = (String(html).match(/"store_show_image"\s*:[^\n]*\n([\s\S]*?)\};\s*console\.log\(datasend\)/) || [])[1] || '';
  const tail = {};
  block.replace(/"([\w]+)"\s*:\s*"([^"]*)"/g, (m, k, v) => { tail[k] = v; return m; });
  if (!tail.id) throw new Error('مقدرتش أقرا صفحة تعديل الصنف من البرنامج - ما اتحفظش حاجة');

  const factory = sel('factory_id');
  const supplier = sel('select_supplier');
  const priceLists = [];
  inputs.forEach(t => {
    const cls = ' ' + (iiAttr_(t, 'class') || '') + ' ';
    if (cls.indexOf(' item_main_max_quantity ') >= 0) priceLists.push({ id: iiAttr_(t, 'data-id'), item_id: iiAttr_(t, 'data-item'), price_list_id: iiAttr_(t, 'data-list'), unit_id: sel('unit_id'), max_quantity: iiAttr_(t, 'value') || '' });
  });
  inputs.forEach(t => {
    const cls = ' ' + (iiAttr_(t, 'class') || '') + ' ';
    if (cls.indexOf(' item_sub_max_quantity ') >= 0) priceLists.push({ id: iiAttr_(t, 'data-id'), item_id: iiAttr_(t, 'data-item'), price_list_id: iiAttr_(t, 'data-list'), unit_id: iiAttr_(t, 'data-unit'), max_quantity: iiAttr_(t, 'value') || '' });
  });
  const v01 = b => b ? 1 : 0;
  const p = {
    item_name: val('item_name'), bolla: val('bolla'), profit_margin: val('profit_margin'), sale_price: val('sale_price'),
    sale_unit: sel('select_type'), quantity: val('quantity'), damaged: val('damaged'),
    factory_id: (factory == 0 || factory === '' || factory == null) ? 0 : factory,
    is_card: v01(checked('is_card')), pay_price: val('pay_price'), about: val('about'), min_quantity: val('mini'),
    max_quantity: val('max_quantity'), max_quantity_gomla: val('max_quantity_gomla'), supplier: JSON.stringify(supplier),
    incom_date: val('incom_date'), expire_date: val('expire_date'), barcode: val('barcode'),
    img: newImg != null ? newImg : oldImg, jsonArray: '[]', deleteArray: '[]', unit_id: sel('unit_id'),
    is_service: v01(checked('is_service')), is_hidden_deficiency_statement: v01(checked('is_hidden_deficiency_statement')),
    cooking_add: v01(checked('cooking_add')), available: v01(checked('available')), require_serial: v01(checked('require_serial')),
    game: v01(checked('game')), is_profit_margin: v01(checked('is_profit_margin')), details: ta != null ? iiDecode_(ta).trim().replace(/\u00a0/g, '&nbsp;') : undefined,
    show_online: radioClass('show_online'), company_id: sel('company_id'), shelf: sel('shelf'), background_color: val('background_color'),
    order_show: val('order_show'), lowest_price: val('lowest_price'), meter_ceramic: val('meter_ceramic'),
    card_company_id: sel('card_company'), warranty_id: sel('warranty_id'), color_list: sel('color_list'), size_list: sel('size_list'),
    items_group_list: sel('items_group_list'), addbarcode: '[]', vat_id: sel('vat_id'), vat_state: radio('vat_state'),
    purchase_vat_id: sel('purchase_vat_id'), purchase_vat_state: radio('purchase_vat_state'),
    price_befor_offer: val('price_befor_offer'), measruing_type: radio('measruing_type'),
    sale_code: val('sale_code'), purchase_code: val('purchase_code'), item_company_id: sel('item_companies'),
    item_price_lists: priceLists, order_unit: (val('order_unit') != null ? val('order_unit') : (sel('order_unit') != null ? sel('order_unit') : 1)),
    store_category_id: sel('store_category_id') !== undefined ? '0' : undefined, store_show_image: v01(checked('store_show_image'))
  };
  Object.keys(tail).forEach(k => { p[k] = tail[k]; });
  return { payload: p, oldImg: oldImg, id: tail.id };
}
// زي jQuery.param (نسخة 3) بالظبط: null/undefined = فاضي، المصفوفات [0][key]، المسافة %20
function iiParam_(obj) {
  const out = [];
  const add = (k, v) => out.push(encodeURIComponent(k) + '=' + encodeURIComponent(v == null ? '' : v));
  const build = (prefix, v) => {
    if (Array.isArray(v)) v.forEach((x, i) => build(prefix + '[' + (typeof x === 'object' && x != null ? i : '') + ']', x));
    else if (v != null && typeof v === 'object') Object.keys(v).forEach(k => build(prefix + '[' + k + ']', v[k]));
    else add(prefix, v);
  };
  Object.keys(obj).forEach(k => build(k, obj[k]));
  return out.join('&');
}
